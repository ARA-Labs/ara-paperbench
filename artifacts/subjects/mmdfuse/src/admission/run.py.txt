"""Attributed orchestration of the unchanged released MMD-FUSE implementation."""
from __future__ import annotations
import hashlib
import json
import os
from pathlib import Path
import sys
import time
from typing import Any

REPETITIONS: int = 200


def main() -> None:
    started = time.monotonic()
    # Keep the original eight-CPU JAX allocation before importing JAX.
    allowed = sorted(os.sched_getaffinity(0))
    source = Path('/source')
    binding = json.loads((Path('/inputs/binding/binding.json')).read_bytes())
    for name, expected in binding['source_hashes'].items():
        if hashlib.sha256((source / name).read_bytes()).hexdigest() != expected:
            raise RuntimeError(f'Author source changed: {name}')
    request = json.loads(Path('/request/request.json').read_bytes())
    values = dict(request['perturbation'])
    if set(values) == {'jax_seed', 'label_permutation'}:
        seed = values['jax_seed']
        if type(seed) is not int or seed not in (42, 43):
            raise ValueError('unregistered seed')
        rows = None
    elif set(values) == {'row_permutation', 'label_permutation'}:
        seed = 42
        identity = values['row_permutation']
        content = Path(f'/inputs/rows/{identity}.json').read_bytes()
        if hashlib.sha256(content).hexdigest() != identity:
            raise ValueError('row manifest identity mismatch')
        rows = json.loads(content)['repetitions']
        if len(rows) != REPETITIONS:
            raise ValueError('missing row manifests')
        for index, row in enumerate(rows):
            if row['repetition'] != index:
                raise ValueError('row identity mismatch')
            for group in ('x', 'y'):
                if any(type(v) is not int for v in row[group]) or sorted(row[group]) != list(range(500)):
                    raise ValueError('not a within-group permutation')
    else:
        raise ValueError('unsupported request fields')
    offset = 0 if rows is None else 8
    selected = allowed[offset:offset+8]
    if len(selected) != 8:
        raise ValueError('registered family requires eight assigned logical CPUs')
    os.sched_setaffinity(0, set(selected))
    label = values['label_permutation']
    if type(label) is not int or label not in (0, 1) or (label and seed != 42):
        raise ValueError('invalid matched label control')
    sys.path.insert(0, '/source')
    import jax
    import jax.numpy as jnp
    from jax import random
    import numpy as np
    from mmdfuse import mmdfuse
    from sampler_mixture import sampler_mixture

    key = random.PRNGKey(seed)
    rejections: list[int] = []
    records: list[dict[str, Any]] = []
    for repetition in range(REPETITIONS):
        key, sample_key = random.split(key)
        x, y = sampler_mixture(sample_key, m=500, n=500, d=2, mu=20, std_1=1, std_2=1.3)
        key, test_key = random.split(key)
        drawn_x = hashlib.sha256(np.asarray(x).tobytes()).hexdigest()
        drawn_y = hashlib.sha256(np.asarray(y).tobytes()).hexdigest()
        if label:
            # An independent fold-in stream leaves the author sampler/test schedule unchanged.
            label_key = random.fold_in(sample_key, 20260913)
            indices = random.permutation(label_key, 1000)
            pooled = jnp.concatenate((x, y), axis=0)
            x, y = pooled[indices[:500]], pooled[indices[500:]]
        elif rows is not None:
            x, y = x[jnp.array(rows[repetition]['x'])], y[jnp.array(rows[repetition]['y'])]
        rejected = int(mmdfuse(x, y, test_key))
        if rejected not in (0, 1):
            raise RuntimeError('nonbinary author test result')
        rejections.append(rejected)
        record: dict[str, Any] = {
            'repetition': repetition, 'sample_key': np.asarray(sample_key).tolist(),
            'test_key': np.asarray(test_key).tolist(), 'drawn_x_sha256': drawn_x,
            'drawn_y_sha256': drawn_y,
            'x_sha256': hashlib.sha256(np.asarray(x).tobytes()).hexdigest(),
            'y_sha256': hashlib.sha256(np.asarray(y).tobytes()).hexdigest(),
            'rejected': rejected, 'elapsed_seconds': time.monotonic() - started,
        }
        records.append(record)
        print(json.dumps(record, sort_keys=True), flush=True)
    result = {'rejection_proportion': sum(rejections)/REPETITIONS,
              'power_author': float(jnp.mean(jnp.array(rejections))),
              'repetitions': REPETITIONS, 'rejections': sum(rejections), 'records': records,
              'elapsed_seconds': time.monotonic()-started, 'jax_version': jax.__version__,
              'source_hashes': binding['source_hashes'], 'request': request,
              'attribution': 'released implementation; lambda_multiplier=1; not paper temperature'}
    with Path('/output/metric.json').open('x') as stream:
        json.dump(result, stream, sort_keys=True)
        stream.flush()
        os.fsync(stream.fileno())

if __name__ == '__main__':
    main()
